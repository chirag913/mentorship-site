<script>
(function(){
  var f = {
    price:  document.getElementById('price'),
    cost:   document.getElementById('cost'),
    ship:   document.getElementById('ship'),
    rship:  document.getElementById('rship'),
    ad:     document.getElementById('ad'),
    gst:    document.getElementById('gst'),
    orders: document.getElementById('orders'),
    rto:    document.getElementById('rto')
  };

  var out = {
    rto:       document.getElementById('rtoOut'),
    gmv:       document.getElementById('gmv'),
    ideal:     document.getElementById('ideal'),
    real:      document.getElementById('real'),
    barKept:   document.getElementById('barKept'),
    verdict:   document.getElementById('verdict'),
    breakeven: document.getElementById('breakeven')
  };

  function num(el, fallback){
    var v = parseFloat(el.value);
    return (isNaN(v) || v < 0) ? fallback : v;
  }

  function inr(n){
    var neg = n < 0;
    n = Math.round(Math.abs(n));
    var s = n.toString();
    if (s.length > 3){
      var last3 = s.slice(-3);
      var rest  = s.slice(0, -3);
      s = rest.replace(/\B(?=(\d{2})+(?!\d))/g, ',') + ',' + last3;
    }
    return (neg ? '−₹' : '₹') + s;
  }

  function calc(){
    var price  = num(f.price, 0),
        cost   = num(f.cost, 0),
        ship   = num(f.ship, 0),
        rship  = num(f.rship, 0),
        ad     = num(f.ad, 0),
        gstPct = Math.min(num(f.gst, 0), 100),
        orders = Math.max(Math.round(num(f.orders, 1)), 1),
        rtoPct = num(f.rto, 0);

    var gstAmt = price * (gstPct / 100);

    // profit on an order that actually gets delivered
    var perDelivered = price - cost - ship - ad - gstAmt;
    // money burnt on an order that comes back
    var perRTO = ad + ship + rship;

    var rto       = rtoPct / 100;
    var delivered = orders * (1 - rto);
    var returned  = orders * rto;

    var idealProfit = orders * perDelivered;
    var realProfit  = (delivered * perDelivered) - (returned * perRTO);
    var gmv         = orders * price;

    // RTO % where profit hits exactly zero
    var beDen = perDelivered + perRTO;
    var breakeven = (perDelivered <= 0) ? 0
                  : (beDen <= 0 ? 100 : (perDelivered / beDen) * 100);

    // paint
    out.rto.textContent   = Math.round(rtoPct);
    out.gmv.textContent   = inr(gmv);
    out.ideal.textContent = inr(idealProfit);
    out.real.textContent  = inr(realProfit);

    var realEl = out.real;
    if (realProfit < 0) realEl.classList.add('neg');
    else realEl.classList.remove('neg');

    var keptShare = (idealProfit > 0) ? (realProfit / idealProfit) * 100 : 0;
    keptShare = Math.max(0, Math.min(100, keptShare));
    out.barKept.style.width = keptShare.toFixed(1) + '%';

    out.breakeven.textContent = Math.max(0, Math.round(breakeven)) + '%';

    // verdict copy
    var v;
    if (perDelivered <= 0){
      v = '<span class="warn">Har delivered order pe bhi loss hai.</span> Pricing ya cost structure pehle theek karo — RTO uske baad ki problem hai.';
    } else if (realProfit < 0){
      v = 'Is RTO rate pe tum <span class="warn">' + inr(Math.abs(realProfit)) + ' ka loss</span> kar rahe ho. Sale ho rahi hai, paisa shipping aur ads mein jaa raha hai.';
    } else {
      var lostPct = Math.round(100 - keptShare);
      var margin  = (gmv > 0) ? (realProfit / gmv) * 100 : 0;
      v = 'RTO ne tumhara <b>' + lostPct + '% profit</b> kha liya. Real margin: <b>' +
          margin.toFixed(1) + '%</b> — jo dikh raha tha ' +
          ((gmv > 0) ? ((idealProfit / gmv) * 100).toFixed(1) : '0') + '% uske against.';
    }
    out.verdict.innerHTML = v;
  }

  Object.keys(f).forEach(function(k){
    f[k].addEventListener('input', calc);
  });

  calc();
})();
</script>
